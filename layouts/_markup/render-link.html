{{- $destination := .Destination -}} {{- $parsedURL := urls.Parse $destination -}} {{- $siteURL := urls.Parse site.BaseURL -}} {{- $isWebURL := or (eq $parsedURL.Scheme "http") (eq $parsedURL.Scheme "https") -}} {{- $isExternal := and $isWebURL (ne $parsedURL.Hostname $siteURL.Hostname) -}} {{- $path := strings.TrimPrefix "/" $parsedURL.Path -}} {{- $isInternalArticle := and (or (not $parsedURL.Hostname) (eq $parsedURL.Hostname $siteURL.Hostname)) (strings.HasPrefix $path "articles/") -}} {{- $opensNewTab := or $isExternal $isInternalArticle -}} {{ .Text | safeHTML }}{{ if $opensNewTab }} (opens in a new tab){{ end }}
